# ATENCAO: o app vive em /opt/avagym. O caminho /opt/acade NAO existe mais e
# /opt/acade-deploy.sh esta quebrado (aponta para o diretorio antigo).
# O repositorio da VPS fica em HEAD destacado, entao "git pull" falha: use fetch + checkout.

cd /opt/avagym
git fetch origin
git checkout --detach origin/main

cd app/api && npm ci --omit=dev && cd ../..
cd app/frontend && npm ci && npm run build && cd ../..

pm2 startOrRestart /opt/avagym-runtime/ecosystem.config.cjs --update-env
pm2 save
pm2 logs avagym-api --lines 20 --nostream
pm2 logs avagym-web --lines 20 --nostream
